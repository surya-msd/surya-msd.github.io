    /*
* Method Name : seasonOpenCheck
* Description : Method to check seasons
* Params      : currSeason,futSeason,granularity,screen.
*/
    public static void seasonOpenCheck(Season__c currSeason, Season__c futSeason, String granularity, String screen){
        system.debug('currSeason=='+currSeason);
        system.debug('futSeason=='+futSeason);
        isAPMet = (System.today() >= currSeason.AP_Process_Open_Date_Submitted__c && System.today() <= currSeason.AP_Process_Close_Date_Submitted__c) && (currSeason.AP_Product_Granularity_Submitted__c == granularity) ? true : false;
        isISFMet = (System.today() >= currSeason.ISF_Process_Open_Date_Submitted__c && System.today() <= currSeason.ISF_Process_Close_Date_Submitted__c) && currSeason.ISF_Product_Granularity_Submitted__c == granularity ? true : false;
        isCy1APMet = (System.today() >= futSeason.AP_Process_Open_Date_Submitted__c && System.today() <= futSeason.AP_Process_Close_Date_Submitted__c) && futSeason.AP_Product_Granularity_Submitted__c == granularity ? true : false;
        
        if(screen == 'Quarterly'){
            isCy1FSFMet = (System.today() >= futSeason.FSF_Process_Open_Date_Submitted__c && System.today() <= futSeason.FSF_Process_Close_Date_Submitted__c) && futSeason.FSF_Product_Granularity_Submitted__c == granularity  && futSeason.FSF_Quaterly_Submitted__c == true ? true : false;
            system.debug('isCy1FSFMet=='+isCy1FSFMet);
            system.debug('isCy1APMet=='+isCy1APMet);
        } else {
            isCy1FSFMet = (System.today() >= futSeason.FSF_Process_Open_Date_Submitted__c && System.today() <= futSeason.FSF_Process_Close_Date_Submitted__c) && futSeason.FSF_Product_Granularity_Submitted__c == granularity && futSeason.FSF_Quaterly_Submitted__c == false ? true : false;
        }
        
    }

    public static void generateSeasonMap(List<Season__c> currSeasons,List<Season__c> futSeasons){
        currSeasonMap = new Map<String,Season__c>();
        futSeasonMap = new Map<String,Season__c>();
        for(Season__c seas:currSeasons) {
            currSeasonMap.put(seas.Product_Group__c,seas);        	
        }
        for(Season__c seas:futSeasons) {
            futSeasonMap.put(seas.Product_Group__c,seas);
        }
    }

    Public static Decimal fetchOpeningInventory(String terrId,Season__c currSeason, String accId, String productId){

        String prevyr = String.valueOf(Integer.valueOf(currSeason.Year__c)-1);
        List<OpportunityLineItemSchedule> previousOLIS = [Select Actual_Inventory_Qty__c from OpportunityLineItemSchedule 
                                                          Where OpportunityLineItem.Product2Id =: productId AND
                                                          OpportunityLineItem.Opportunity.Territory2Id =: terrId AND
                                                          OpportunityLineItem.Opportunity.AccountId =: accId AND
                                                          OpportunityLineItem.Opportunity.Season_SFID__r.Season_Cd__c =: currSeason.Season_Cd__c AND
                                                          OpportunityLineItem.Opportunity.Season_SFID__r.Year__c =: prevyr AND
                                                          OpportunityLineItem.Opportunity.Product_Group__c =: currSeason.Product_Group__c AND Line_Type__c = 'Monthly' ORDER BY Year__c,fCalendarMonth__c DESC LIMIT 1];
        if(!previousOLIS.isEmpty()){
            OpenInv = previousOLIS[0].Actual_Inventory_Qty__c != null ? previousOLIS[0].Actual_Inventory_Qty__c : 0;
        }
        return OpenInv; 
    }
